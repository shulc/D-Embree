cmake_minimum_required(VERSION 3.24)
project(dembree VERSION 0.1.0 LANGUAGES C CXX)
include(FetchContent)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/.cmake/api/v1/query")
file(WRITE "${CMAKE_BINARY_DIR}/.cmake/api/v1/query/codemodel-v2" "")
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE STREQUAL "Release")
  message(FATAL_ERROR "Only Release scalar traversal has been qualified")
endif()
option(DEMBREE_TEST_FAULTS "Enable deterministic test hooks (never production)" OFF)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/native/inputs.lock.json" lock)
string(JSON upstream_url GET "${lock}" url)
string(JSON upstream_sha GET "${lock}" sha256)
set(EMBREE_STATIC_LIB ON CACHE BOOL "" FORCE)
set(EMBREE_TASKING_SYSTEM INTERNAL CACHE STRING "" FORCE)
set(EMBREE_ISPC_SUPPORT OFF CACHE BOOL "" FORCE)
set(EMBREE_TUTORIALS OFF CACHE BOOL "" FORCE)
set(EMBREE_SYCL_SUPPORT OFF CACHE BOOL "" FORCE)
set(EMBREE_FILTER_FUNCTION ON CACHE BOOL "" FORCE)
set(EMBREE_GEOMETRY_USER ON CACHE BOOL "" FORCE)
foreach(kind TRIANGLE QUAD CURVE SUBDIVISION INSTANCE INSTANCE_ARRAY GRID POINT)
  set(EMBREE_GEOMETRY_${kind} OFF CACHE BOOL "" FORCE)
endforeach()
set(EMBREE_MAX_INSTANCE_LEVEL_COUNT 1 CACHE STRING "" FORCE)
set(EMBREE_RAY_PACKETS OFF CACHE BOOL "" FORCE)
set(EMBREE_IGNORE_INVALID_RAYS OFF CACHE BOOL "" FORCE)
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" dembree_processor)
if(dembree_processor MATCHES "^(aarch64|arm64)$")
  set(EMBREE_MAX_ISA NEON CACHE STRING "" FORCE)
else()
  set(EMBREE_MAX_ISA SSE2 CACHE STRING "" FORCE)
endif()
if(EMSCRIPTEN)
  set(FLAGS_SSE2 "-msse -msse2" CACHE STRING "Emscripten SSE2 spelling" FORCE)
endif()
# Both original archive and checked patch belong to this build root.
FetchContent_Declare(dembree_upstream URL "${upstream_url}" URL_HASH "SHA256=${upstream_sha}"
  SOURCE_DIR "${CMAKE_BINARY_DIR}/s"
  BINARY_DIR "${CMAKE_BINARY_DIR}/b"
  SUBBUILD_DIR "${CMAKE_BINARY_DIR}/p"
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE
  PATCH_COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/apply_patch.py" <SOURCE_DIR>)
FetchContent_MakeAvailable(dembree_upstream)
# Also validate reconfiguration, for which FetchContent can skip PATCH_COMMAND.
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/apply_patch.py" "${dembree_upstream_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
target_compile_definitions(embree PRIVATE DEMBREE_CLOSED_INTERVAL=1)
if(MSVC)
  target_compile_options(embree PRIVATE /fp:strict)
else()
  target_compile_options(embree PRIVATE -fno-fast-math -ffp-contract=off)
endif()
if(EMSCRIPTEN)
  add_library(dembree STATIC csrc/embree_c.cpp)
  target_compile_options(dembree PRIVATE -fexceptions -pthread -msimd128)
  target_link_options(dembree INTERFACE -pthread -msimd128 -fexceptions -sDISABLE_EXCEPTION_CATCHING=0 -sPTHREAD_POOL_SIZE=2)
else()
  add_library(dembree SHARED csrc/embree_c.cpp)
  set_target_properties(dembree PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  if(APPLE)
    set_target_properties(dembree PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR YES)
    target_link_options(dembree PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/csrc/exports.apple")
  elseif(UNIX)
    target_link_options(dembree PRIVATE "LINKER:--exclude-libs,ALL" "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/csrc/exports.map")
  endif()
endif()
target_compile_features(dembree PRIVATE cxx_std_17)
target_compile_definitions(dembree PRIVATE DEMBREE_BUILD)
target_include_directories(dembree PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(dembree PRIVATE embree)
if(MSVC)
  target_compile_options(dembree PRIVATE /fp:strict)
else()
  target_compile_options(dembree PRIVATE -fno-fast-math -ffp-contract=off -fstack-usage)
endif()
if(DEMBREE_TEST_FAULTS)
  target_compile_definitions(dembree PRIVATE DEMBREE_TEST_FAULTS)
endif()
install(TARGETS dembree RUNTIME DESTINATION bin COMPONENT runtime LIBRARY DESTINATION lib COMPONENT runtime ARCHIVE DESTINATION lib COMPONENT runtime)
install(FILES include/embree_c.h native/inputs.lock.json DESTINATION include/dembree COMPONENT runtime)
install(FILES LICENSE THIRD_PARTY_LICENSES.md "${dembree_upstream_SOURCE_DIR}/LICENSE.txt" "${dembree_upstream_SOURCE_DIR}/third-party-programs.txt" DESTINATION share/licenses/dembree COMPONENT runtime)
if(EXISTS "${dembree_upstream_SOURCE_DIR}/NOTICE")
  install(FILES "${dembree_upstream_SOURCE_DIR}/NOTICE" DESTINATION share/licenses/dembree COMPONENT runtime)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests/windows_macros)
  add_executable(dembree_smoke tests/native/smoke.cpp)
  target_compile_features(dembree_smoke PRIVATE cxx_std_17)
  target_include_directories(dembree_smoke PRIVATE tests/fixtures csrc)
  target_link_libraries(dembree_smoke PRIVATE dembree)
  if(DEMBREE_TEST_FAULTS)
    target_compile_definitions(dembree_smoke PRIVATE DEMBREE_TEST_FAULTS)
  endif()
  if(EMSCRIPTEN)
    target_link_libraries(dembree_smoke PRIVATE embree)
    target_link_options(dembree_smoke PRIVATE -sEXIT_RUNTIME=1 -sENVIRONMENT=node,web -sSTACK_SIZE=1048576)
  else()
    if(APPLE)
      set_target_properties(dembree_smoke PROPERTIES INSTALL_RPATH "@loader_path/../lib")
    else()
      set_target_properties(dembree_smoke PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
    endif()
    install(TARGETS dembree_smoke RUNTIME DESTINATION bin COMPONENT smoke)
  endif()
  add_test(NAME dembree_smoke COMMAND dembree_smoke)
endif()
if(BUILD_TESTING)
  add_executable(dembree_exact tests/native/exact.cpp)
  target_compile_features(dembree_exact PRIVATE cxx_std_17)
  target_include_directories(dembree_exact PRIVATE tests/fixtures csrc)
  target_link_libraries(dembree_exact PRIVATE dembree)
  if(DEMBREE_TEST_FAULTS)
    target_compile_definitions(dembree_exact PRIVATE DEMBREE_TEST_FAULTS)
  endif()
  if(EMSCRIPTEN)
    target_link_libraries(dembree_exact PRIVATE embree)
    target_link_options(dembree_exact PRIVATE -sEXIT_RUNTIME=1 -sENVIRONMENT=node,web -sSTACK_SIZE=1048576)
  endif()
  add_test(NAME dembree_exact COMMAND dembree_exact)
endif()
